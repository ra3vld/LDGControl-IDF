cmake_minimum_required(VERSION 3.22)
set(LDG_BOARD "sunton" CACHE STRING "headless, sunton or guition")
if(NOT LDG_BOARD MATCHES "^(headless|sunton|guition)$")
  message(FATAL_ERROR "Unknown LDG_BOARD")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_SOURCE_DIR}/boards/common.defaults")
if(LDG_BOARD STREQUAL "guition")
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_SOURCE_DIR}/boards/guition.defaults")
  set(IDF_TARGET esp32s3)
else()
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_SOURCE_DIR}/boards/esp32.defaults")
  set(IDF_TARGET esp32)
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
# SDK configure passes may rewrite registry packages as local managed paths.
# Keep that generated state in the build tree, preserving the pinned contract.
configure_file("${CMAKE_SOURCE_DIR}/dependencies.${IDF_TARGET}.lock"
    "${CMAKE_BINARY_DIR}/dependencies.lock" COPYONLY)
idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_BINARY_DIR}/dependencies.lock")
project(ldgcontrol VERSION 0.1.0)
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" VERSION_EQUAL "5.5.5")
  message(FATAL_ERROR "ESP-IDF 5.5.5 is required")
endif()
